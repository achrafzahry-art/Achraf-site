// Sales estimate by store × brand × country, in EUR
const estimate = (row, drv, fx) => {
  const d = drv[row.country][row.brand];
  return row.lastYear                       // same month, last year
    * (1 + d.likeForLike)                   // LFL growth by market
    * d.seasonality[row.month]              // monthly seasonality
    * (1 + (d.storeUplift[row.store] ?? 0)) // openings, renovations
    * fx[row.country];                      // local currency → EUR
};

const rows = sales.map(r => ({ ...r, est: estimate(r, drivers, fx) }));

// Roll up by country and brand, rank by gap to budget
const byMarket = Object.groupBy(rows, r => `${r.country} · ${r.brand}`);

const view = Object.entries(byMarket).map(([market, rs]) => {
  const est    = rs.reduce((s, r) => s + r.est, 0);
  const budget = rs.reduce((s, r) => s + r.budget, 0);
  return { market, est, budget, gapPct: est / budget - 1 };
}).sort((a, b) => Math.abs(b.gapPct) - Math.abs(a.gapPct));

console.table(view);
